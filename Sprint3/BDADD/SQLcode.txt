-- US23 - As a Production Manager, I want to make sure that the expected execution time of an operation is not greater than the maximum execution time of every workstation type where it may be run.

CREATE OR REPLACE TRIGGER trgExecutionTimeGreaterMax
BEFORE INSERT OR UPDATE OF ExecutionTime -- Trigger activates before inserting or updating the ExecutionTime column
ON Operation -- The trigger is defined on the Operation table
FOR EACH ROW -- Trigger operates at the row level
DECLARE
    v_max_time WorkstationTypes_Operation_Type.MaxTime%TYPE; -- Variable to store the maximum allowed execution time
    ex_time_exceeded EXCEPTION; -- Custom exception to handle cases where execution time exceeds the maximum allowed
BEGIN
    -- Fetch the maximum allowed execution time for the given operation type (OP_TYPE)
    SELECT MAX(wt.MaxTime)
    INTO v_max_time
    FROM WorkstationTypes_Operation_Type wt
    WHERE wt.OP_TYPE = :NEW.OP_TYPE; -- Match the operation type of the new row being inserted/updated

    -- Check if the new execution time exceeds the maximum allowed time
    IF :NEW.ExecutionTime > v_max_time THEN
        RAISE ex_time_exceeded; -- Raise a custom exception if execution time exceeds the limit
    END IF;

EXCEPTION
    -- Handle the custom exception when execution time exceeds the maximum allowed
    WHEN ex_time_exceeded THEN
        RAISE_APPLICATION_ERROR(
            -20101, -- Custom error code
            'Execution time of ' || :NEW.ExecutionTime || -- Error message includes the invalid execution time
            ' exceeds the maximum allowed time of ' || v_max_time || -- Includes the maximum allowed time
            ' for operation type ' || :NEW.OP_TYPE || -- Includes the operation type
            '. Please provide a valid ExecutionTime.' -- Suggests providing a valid execution time
        );
END;
/


-- US24 -  As a Production Manager, I don’t want it to be possible to add a product as an input in its own BOO.
    
CREATE OR REPLACE TRIGGER add_product_input_in_own_BOO
BEFORE INSERT OR UPDATE OF PartNumber -- Trigger activates before inserting or updating the PartNumber column
ON BOO_INPUT -- The trigger is defined on the BOO_INPUT table
FOR EACH ROW -- Trigger operates at the row level
DECLARE
    product_id Operation.ProductID%TYPE; -- Variable to store the ProductID of the product associated with the operation
    circular_error EXCEPTION; -- Custom exception to handle circular references
BEGIN
    -- Fetch the ProductID for the operation being referenced
    SELECT O.ProductID
    INTO product_id
    FROM Operation O
    WHERE :NEW.OPID = O.OPID; -- Match the operation ID (OPID) of the new row being inserted/updated

    -- Check if the product is being referenced as an input in its own BOO
    IF product_id = :NEW.PartNumber THEN
        RAISE circular_error; -- Raise a custom exception for circular reference
    END IF;

EXCEPTION
    -- Handle the custom exception for circular references
    WHEN circular_error THEN
        RAISE_APPLICATION_ERROR(
            -20002, -- Custom error code
            'A product cannot be an input in its own BOO.' -- Error message describing the issue
        );
END;
/

-- USB25 - As a Production Manager, I want to get a list of a product’s operations necessary for the production plan module.

-- Function: GetProductIDs
-- Purpose: Retrieve all product IDs, including subproducts, recursively for a given product ID
CREATE OR REPLACE FUNCTION GetProductIDs(p_product_id IN Product.ProductID%TYPE)
RETURN SYS_REFCURSOR
IS
    result_cursor SYS_REFCURSOR; -- Variable to hold the reference cursor
BEGIN
    -- Open a cursor to execute the recursive query
    OPEN result_cursor FOR
    WITH ProductIDs (ProductID) AS ( -- Recursive Common Table Expression (CTE)
        -- Base case: Select the given product's operations
        SELECT o.ProductID
        FROM Operation o
        WHERE o.ProductID = p_product_id
        
        UNION ALL -- Combine results with the recursive part
        
        -- Recursive part: Select subproduct IDs referenced in the BOO_INPUT
        SELECT BI.PartNumber
        FROM BOO_INPUT BI
        JOIN Operation O ON O.OPID = BI.OPID -- Join BOO_INPUT with Operation table
        JOIN ProductIDs pi ON pi.ProductID = O.ProductID -- Recursive join with ProductIDs
        WHERE BI.PartNumber IN ( -- Filter to ensure valid product IDs
                SELECT o.ProductID
                FROM Operation o
           )
    )
    -- Retrieve distinct product IDs from the CTE
    SELECT DISTINCT ProductID
    FROM ProductIDs;
    
    RETURN result_cursor; -- Return the cursor with the result set
END GetProductIDs;
/

-- Function: GetProductOperationIDs
-- Purpose: Retrieve all operation IDs and their execution times for a given product ID
CREATE OR REPLACE FUNCTION GetProductOperationIDs (p_product_id IN Product.ProductID%TYPE)
RETURN SYS_REFCURSOR
IS
    result_cursor SYS_REFCURSOR; -- Variable to hold the reference cursor
BEGIN
    -- Open a cursor to execute the query for operation IDs and execution times
    OPEN result_cursor FOR
        SELECT o.OPID, o.ExecutionTime -- Select operation ID and execution time
        FROM Operation o
        WHERE o.ProductID = p_product_id; -- Filter by the given product ID
    RETURN result_cursor; -- Return the cursor with the result set
END;
/

-- Anonymous PL/SQL Block
-- Purpose: Retrieve and print product IDs, operations, and their inputs/outputs
DECLARE
    operations SYS_REFCURSOR; -- Cursor to hold operations for a product
    products_ids SYS_REFCURSOR; -- Cursor to hold product IDs
    op_id  BOO_INPUT.OPID%TYPE; -- Variable to store the operation ID
    execution_time NUMBER; -- Variable to store the execution time of an operation
    p_id Product.ProductID%TYPE; -- Variable to store product IDs
    input_partnumber BOO_INPUT.PartNumber%TYPE; -- Variable to store input part numbers
    output_partnumber BOO_OUTPUT.PartNumber%TYPE; -- Variable to store output part numbers
BEGIN
    -- Call the GetProductIDs function with the initial product ID ('AS12946S20')
    products_ids := GetProductIDs('AS12946S20');
    
    -- Loop through the cursor for product IDs
    LOOP
        FETCH products_ids INTO p_id; -- Fetch the next product ID
        EXIT WHEN products_ids%NOTFOUND; -- Exit loop when no more product IDs

        -- Call the GetProductOperationIDs function for the current product ID
        operations := GetProductOperationIDs(p_id);

        -- Loop through the cursor for operations
        LOOP
            FETCH operations INTO op_id, execution_time; -- Fetch operation ID and execution time
            EXIT WHEN operations%NOTFOUND; -- Exit loop when no more operations

            -- Print the operation ID and execution time
            DBMS_OUTPUT.PUT_LINE('Operation ID: ' || op_id || ', Execution Time: ' || execution_time);

            -- Retrieve and print inputs related to the operation
            FOR input IN (
                SELECT PartNumber -- Select input part numbers
                FROM BOO_INPUT
                WHERE OPID = op_id -- Filter by the current operation ID
            ) LOOP
                input_partnumber := input.PartNumber; -- Assign the input part number
                DBMS_OUTPUT.PUT_LINE('Input: ' || input_partnumber); -- Print the input
            END LOOP;

            -- Retrieve and print outputs related to the operation
            FOR output IN (
                SELECT PartNumber -- Select output part numbers
                FROM BOO_OUTPUT
                WHERE OPID = op_id -- Filter by the current operation ID
            ) LOOP
                output_partnumber := output.PartNumber; -- Assign the output part number
                DBMS_OUTPUT.PUT_LINE('Output: ' || output_partnumber); -- Print the output
            END LOOP;

        END LOOP;

        -- Close the cursor for operations
        CLOSE operations;
    END LOOP;

    -- Close the cursor for product IDs
    CLOSE products_ids;
END;
/

    v_stock NUMBER; -- Variable to store the stock level
    v_min_stock NUMBER; -- Variable to store the minimum stock level
    v_reserved NUMBER := 0; -- Initialize reserved stock to 0
    v_Part_Type Part.PartType%TYPE; -- Variable to store the part type
BEGIN
    -- Retrieve the part type for the given part number
    SELECT PartType
    INTO v_Part_Type
    FROM Part
    WHERE PartNumber = p_PartNumber;

    -- Retrieve the reserved quantity; default to 0 if not found
    BEGIN
        SELECT NVL(Quantity, 0)
        INTO v_reserved
        FROM ReserveList
        WHERE PartNumber = p_PartNumber;
    EXCEPTION
        WHEN NO_DATA_FOUND THEN
            v_reserved := 0;
    END;

    -- Check stock levels based on the part type
    IF v_Part_Type = 'Component' THEN
        -- Retrieve stock and minimum stock for components
        SELECT Stock, Min_stock
        INTO v_stock, v_min_stock
        FROM Component
        WHERE PartNumber = p_PartNumber;

        -- Return TRUE if sufficient stock exists, FALSE otherwise
        RETURN (v_stock - p_Quantity >= (v_reserved + v_min_stock));

    ELSIF v_Part_Type = 'Raw_Material' THEN
        -- Retrieve stock and minimum stock for raw materials
        SELECT Stock, Min_stock
        INTO v_stock, v_min_stock
        FROM Raw_Material
        WHERE PartNumber = p_PartNumber;

        -- Return TRUE if sufficient stock exists, FALSE otherwise
        RETURN (v_stock - p_Quantity >= (v_reserved + v_min_stock));
    ELSE
        -- Return FALSE if the part type is invalid
        RETURN FALSE;
    END IF;
END;

-- US27 -As a Production Manager, I want to reserve the necessary materials and components to fulfil a given order.

-- Function: reserve_order
-- Purpose: Reserve a specific quantity of a part in the ReserveList table
CREATE OR REPLACE FUNCTION reserve_order(
    p_PartNumber IN Part.PartNumber%TYPE, -- Part number to reserve
    p_Quantity IN NUMBER -- Quantity to reserve
)
RETURN BOOLEAN
IS
BEGIN
    -- Attempt to update or insert a reservation in the ReserveList table
    BEGIN
        -- Update an existing reservation
        UPDATE ReserveList
        SET Quantity = Quantity + p_Quantity -- Add the new quantity to the existing reservation
        WHERE PartNumber = p_PartNumber;

        -- Check if the update did not affect any rows
        IF SQL%ROWCOUNT = 0 THEN
            -- If no rows were updated, insert a new reservation
            INSERT INTO ReserveList (PartNumber, Quantity)
            VALUES (p_PartNumber, p_Quantity);
        END IF;

        RETURN TRUE; -- Reservation successfully created or updated
    EXCEPTION
        WHEN OTHERS THEN
            RETURN FALSE; -- Return FALSE in case of any errors
    END;
END;

-- Anonymous PL/SQL Block
-- Purpose: Validate stock, reserve necessary parts, and check if an order can be fulfilled
DECLARE
    operations SYS_REFCURSOR; -- Cursor for operations related to a product
    part_stock SYS_REFCURSOR; -- Cursor for parts needed for an operation
    products_ids SYS_REFCURSOR; -- Cursor for products in a specific order
    p_quantity NUMBER; -- Variable to store the quantity of products ordered
    execution_time NUMBER; -- Variable to store execution time of operations
    op_id BOO_INPUT.OPID%TYPE; -- Variable to store operation IDs
    p_id Product.ProductID%TYPE; -- Variable to store product IDs
    part_id Part.PartNumber%TYPE; -- Variable to store part IDs
    Quantity NUMBER; -- Variable to store part quantities
    v_not_exists NUMBER := 0; -- Flag to track if any part is unavailable or reservation fails
BEGIN
    -- Retrieve the products and their quantities for a specific order (Order ID = 8)
    products_ids := GetOrderProducts(9);

    -- Loop through all products in the order
    LOOP
        FETCH products_ids INTO p_id, p_quantity; -- Fetch product ID and quantity
        EXIT WHEN products_ids%NOTFOUND; -- Exit the loop when no more products

        -- Retrieve operations for the current product
        operations := GetProductOperationIDs(p_id);

        -- Loop through all operations for the current product
        LOOP
            FETCH operations INTO op_id, execution_time; -- Fetch operation ID and execution time
            EXIT WHEN operations%NOTFOUND; -- Exit the loop when no more operations

            -- Retrieve parts and their quantities for the current operation
            part_stock := GetProductParts(op_id);

            -- Loop through all parts required for the current operation
            LOOP
                FETCH part_stock INTO part_id, Quantity; -- Fetch part ID and quantity
                EXIT WHEN part_stock%NOTFOUND; -- Exit the loop when no more parts

                -- Multiply the quantity required by the order factor and check stock
                IF NOT CheckStock(part_id, Quantity * p_quantity) THEN
                    v_not_exists := 1; -- Set flag if stock is insufficient
                    DBMS_OUTPUT.PUT_LINE('Stock Check Failed for Part: ' || part_id);
                    EXIT; -- Exit the loop for parts
                ELSE
                    -- Reserve the part directly after verifying stock
                    IF reserve_order(part_id, Quantity * p_quantity) THEN
                        DBMS_OUTPUT.PUT_LINE('Reserved Part: ' || part_id || ', Quantity: ' || Quantity * p_quantity);
                    ELSE
                        DBMS_OUTPUT.PUT_LINE('Failed to reserve Part: ' || part_id);
                        v_not_exists := 1; -- Set flag if reservation fails
                        EXIT; -- Exit the loop for parts
                    END IF;
                END IF;
            END LOOP;

            -- Exit operation loop if stock or reservation fails
            IF v_not_exists = 1 THEN
                EXIT;
            END IF;

            CLOSE part_stock; -- Close the cursor for parts
        END LOOP;

        -- Exit product loop if stock or reservation fails
        IF v_not_exists = 1 THEN
            EXIT;
        END IF;

        CLOSE operations; -- Close the cursor for operations
    END LOOP;

    CLOSE products_ids; -- Close the cursor for products

    -- Final validation message
    IF v_not_exists != 0 THEN
        DBMS_OUTPUT.PUT_LINE('Cannot reserve the necessary materials and components to fulfil the order');
    ELSE
        DBMS_OUTPUT.PUT_LINE('All necessary materials and components successfully reserved to fulfil the order');
    END IF;
END;

-- USBD30 - As a Factory Manager, I want to use/consume a material/component, i.e. to deduct a given amount from the stock.

-- Function: update_stock
-- Purpose: Updates the stock quantity of a given part (either Component or Raw Material) by reducing the specified amount. Ensures the operation respects the reserved and minimum stock levels.
CREATE OR REPLACE FUNCTION update_stock (
    p_PartNumber IN Part.PartNumber%TYPE, -- Input parameter: Part number to update
    p_Amount IN NUMBER -- Input parameter: Amount to deduct from stock
)
RETURN VARCHAR2
IS
    v_Part_Type VARCHAR2(50); -- Variable to store the type of the part (Component or Raw Material)
    v_reserved NUMBER; -- Variable to store the reserved quantity for the part
    v_stock NUMBER; -- Variable to store the available stock
    v_min_stock NUMBER; -- Variable to store the minimum stock level

    -- Custom exceptions
    stock_insuficient EXCEPTION; -- Raised when stock is insufficient
    not_cm EXCEPTION; -- Raised when the part is neither Component nor Raw Material

BEGIN
    -- Start a transaction savepoint
    SAVEPOINT start_point;

    BEGIN
        -- Fetch the type of the part
        SELECT PartType
        INTO v_Part_Type
        FROM Part
        WHERE PartNumber = p_PartNumber;

        -- Fetch the reserved quantity for the part
        BEGIN
            -- If no reservation exists, treat it as 0
            SELECT NVL(Quantity, 0)
            INTO v_reserved
            FROM ReserveList
            WHERE PartNumber = p_PartNumber;
        EXCEPTION
            WHEN NO_DATA_FOUND THEN
                -- If no data is found in ReserveList, set v_reserved to 0
                v_reserved := 0;
        END;

        -- Check if the part is a Component
        IF v_Part_Type = 'Component' THEN
            -- Fetch stock and minimum stock for the Component
            SELECT Stock, Min_stock
            INTO v_stock, v_min_stock
            FROM Component
            WHERE PartNumber = p_PartNumber;

            -- Verify if stock is sufficient
            IF v_stock - p_Amount >= (v_reserved + v_min_stock) THEN
                -- Update the stock for the Component
                UPDATE Component
                SET Stock = Stock - p_Amount
                WHERE PartNumber = p_PartNumber;
            ELSE
                RAISE stock_insuficient; -- Raise exception for insufficient stock
            END IF;

        -- Check if the part is a Raw Material
        ELSIF v_Part_Type = 'Raw_Material' THEN
            -- Fetch stock and minimum stock for the Raw Material
            SELECT Stock, Min_stock
            INTO v_stock, v_min_stock
            FROM Raw_Material
            WHERE PartNumber = p_PartNumber;

            -- Verify if stock is sufficient
            IF v_stock - p_Amount >= (v_reserved + v_min_stock) THEN
                -- Update the stock for the Raw Material
                UPDATE Raw_Material
                SET Stock = Stock - p_Amount
                WHERE PartNumber = p_PartNumber;
            ELSE
                RAISE stock_insuficient; -- Raise exception for insufficient stock
            END IF;

        ELSE
            -- Raise exception if the part is neither Component nor Raw Material
            RAISE not_cm;
        END IF;

        -- Commit the transaction
        COMMIT;

        -- Return a success message
        RETURN 'Stock update successful';

    EXCEPTION
        -- Handle insufficient stock exception
        WHEN stock_insuficient THEN
            ROLLBACK TO start_point; -- Rollback to the savepoint
            RAISE_APPLICATION_ERROR(-20001, 'Insufficient stock to complete the operation.');

        -- Handle invalid part type exception
        WHEN not_cm THEN
            ROLLBACK TO start_point; -- Rollback to the savepoint
            RAISE_APPLICATION_ERROR(-20002, 'The part is neither a Component nor Raw Material.');

        -- Handle any other exceptions
        WHEN OTHERS THEN
            ROLLBACK TO start_point; -- Rollback to the savepoint
            RETURN 'Error: ' || SQLERRM; -- Return the error message
    END;

END;
/
