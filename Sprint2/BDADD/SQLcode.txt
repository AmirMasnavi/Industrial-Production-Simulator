CREATE TABLE BOO (
  ProductID varchar2(255) NOT NULL, 
  PRIMARY KEY (ProductID));

CREATE TABLE BOO_INPUT (
  PartNumber varchar2(255) NOT NULL, 
  OPID       number(10) NOT NULL, 
  Quantity   number(10) NOT NULL, 
  Unit       varchar2(255) NOT NULL, 
  PRIMARY KEY (PartNumber, 
  OPID));

CREATE TABLE BOO_OUTPUT (
  PartNumber varchar2(255) NOT NULL, 
  OPID       number(10) NOT NULL, 
  Quantity   number(10) NOT NULL, 
  Unit       varchar2(255) NOT NULL, 
  PRIMARY KEY (PartNumber, 
  OPID));

CREATE TABLE Client (
  ClientID varchar2(255) NOT NULL, 
  Name     varchar2(50) NOT NULL, 
  VATIN    varchar2(20) NOT NULL, 
  Address  varchar2(100) NOT NULL, 
  ZIP      varchar2(20) NOT NULL, 
  Town     varchar2(50) NOT NULL, 
  Country  varchar2(20) NOT NULL, 
  Email    varchar2(100) NOT NULL, 
  Phone    number(15) NOT NULL, 
  Status   varchar2(255) NOT NULL, 
  PRIMARY KEY (ClientID));

CREATE TABLE Components (
  PartNumber varchar2(255) NOT NULL, 
  PRIMARY KEY (PartNumber));

CREATE TABLE Operation (
  OPID      number(10) NOT NULL, 
  ProductID varchar2(255) NOT NULL, 
  OP_TYPE   number(10) NOT NULL, 
  Next_OP   number(10), 
  PRIMARY KEY (OPID));

CREATE TABLE Operation_Type (
  OP_TYPE     number(10) NOT NULL, 
  Description varchar2(255) NOT NULL, 
  PRIMARY KEY (OP_TYPE));

CREATE TABLE Orders (
  OrderID      varchar2(255) NOT NULL, 
  DateOrder    date NOT NULL, 
  DateDelivery date NOT NULL, 
  Status       varchar2(255) NOT NULL, 
  ClientID     varchar2(255) NOT NULL, 
  PRIMARY KEY (OrderID));

CREATE TABLE Part (
  PartNumber  varchar2(255) NOT NULL, 
  Description varchar2(200) NOT NULL, 
  PRIMARY KEY (PartNumber));

CREATE TABLE Product (
  ProductID      varchar2(255) NOT NULL, 
  FamilyID       number(10) NOT NULL, 
  Name           varchar2(20) NOT NULL, 
  Description    varchar2(200) NOT NULL, 
  PartNumber varchar2(255), 
  PRIMARY KEY (ProductID));

CREATE TABLE Product_Family (
  FamilyID number(10) NOT NULL, 
  Name     varchar2(20) NOT NULL, 
  PRIMARY KEY (FamilyID));

CREATE TABLE Product_Orders (
  OrderID          varchar2(255) NOT NULL, 
  ProductID        varchar2(255) NOT NULL, 
  QuantityProducts number(10) NOT NULL, 
  PRIMARY KEY (OrderID, 
  ProductID));

CREATE TABLE ProductionOrder (
  PO_ID     nvarchar2(255) NOT NULL, 
  StartDate date NOT NULL, 
  EndDate   date NOT NULL, 
  Sequence  varchar2(100) NOT NULL, 
  OrderID   varchar2(255) NOT NULL, 
  ProductID varchar2(255) NOT NULL, 
  PRIMARY KEY (PO_ID));

CREATE TABLE RawMaterial (
  PartNumber varchar2(255) NOT NULL, 
  PRIMARY KEY (PartNumber));

CREATE TABLE Workstations (
  WSID        varchar2(255) NOT NULL, 
  Name        varchar2(20) NOT NULL, 
  Description varchar2(300) NOT NULL, 
  WTID        varchar2(255) NOT NULL, 
  PRIMARY KEY (WSID));
COMMENT ON COLUMN Workstations.Name IS 'minutes';

CREATE TABLE Workstations_ProductionOrder (
  WSID  varchar2(255) NOT NULL, 
  PO_ID nvarchar2(255) NOT NULL, 
  PRIMARY KEY (WSID, 
  PO_ID));

CREATE TABLE WorkstationTypes (
  WTID varchar2(255) NOT NULL, 
  Name varchar2(100) NOT NULL, 
  PRIMARY KEY (WTID));

CREATE TABLE WorkstationTypes_Operation_Type (
  WTID    varchar2(255) NOT NULL, 
  OP_TYPE number(10) NOT NULL, 
  PRIMARY KEY (WTID, 
  OP_TYPE));

------- USBD12 -  As a Production Manager, I want to get the list of parts used in a product.

CREATE OR REPLACE FUNCTION list_parts_used_product(
    product_id IN Operation.ProductID%TYPE -- Input product ID
)
RETURN SYS_REFCURSOR -- Returns a cursor with the list of parts
IS
    list_parts_cursor SYS_REFCURSOR; -- Cursor to hold the result
BEGIN
    -- Query to list parts directly associated with the product and its subassemblies
    OPEN list_parts_cursor FOR
        -- Parts directly associated with the product
        SELECT BI.PartNumber, 
               SUM(BI.Quantity) AS Quantity
        FROM BOO_INPUT BI
        JOIN Operation O ON BI.OPID = O.OPID
        WHERE O.ProductID = product_id
        GROUP BY BI.PartNumber
        
        UNION ALL

        -- Parts associated with subassemblies (recursive case)
        SELECT BO.PartNumber, 
               SUM(BO.Quantity) AS Quantity
        FROM BOO_OUTPUT BO
        JOIN Operation O ON BO.OPID = O.OPID
        JOIN BOO_INPUT BI ON BI.OPID = BO.OPID
        WHERE BO.PartNumber NOT LIKE O.ProductID
          AND O.ProductID IN (
              SELECT BI.PartNumber
              FROM BOO_INPUT BI
              JOIN Part P ON BI.PartNumber = P.PartNumber
              JOIN Operation O ON BI.OPID = O.OPID
              WHERE O.ProductID = product_id 
              GROUP BY BI.PartNumber
          )
        GROUP BY BO.PartNumber;

    -- Return the cursor with the results
    RETURN list_parts_cursor;
END;
/

-- Block to test the function
DECLARE
    list_parts_cursor   SYS_REFCURSOR; -- Cursor to hold the result
    part_number         BOO_INPUT.PartNumber%TYPE; -- Variable to store part number
    quantity            BOO_INPUT.Quantity%TYPE;  -- Variable to store quantity
    product_id          VARCHAR2(20) := 'AS12946S22'; -- Product ID for testing
BEGIN
    -- Display the product ID
    DBMS_OUTPUT.PUT_LINE('List of parts for product: ' || product_id);
    
    -- Call the function to get parts associated with the product
    list_parts_cursor := list_parts_used_product(product_id);
    
    -- Loop through the cursor results
    LOOP
        FETCH list_parts_cursor INTO part_number, quantity;
        EXIT WHEN list_parts_cursor%NOTFOUND;
        
        -- Display the part number and quantity
        DBMS_OUTPUT.PUT_LINE('Part Number: ' || part_number || ', Quantity: ' || quantity);
    END LOOP;
    
    -- Close the cursor after processing
    CLOSE list_parts_cursor;
END;
/


------- USBD13 - As a Production Manager, I want to get a list of operations involved in the production of a product, as well as each workstation type.

CREATE OR REPLACE FUNCTION list_operations_and_workstationTypes(
    product_id IN Product.ProductID%TYPE -- Input parameter for the product ID
)
RETURN SYS_REFCURSOR -- Returns a cursor with the list of operations and workstation types
AS
    operation_workstationTypes_cursor SYS_REFCURSOR; -- Declare the cursor to store the result
BEGIN
    -- Open the cursor to execute the query
    OPEN operation_workstationTypes_cursor FOR
        -- First query: Operations directly associated with the product
        SELECT O.OPID, WtOp.WTID -- Operation ID and Workstation Type ID
        FROM Operation O
        JOIN WorkstationTypes_Operation_Type WtOp ON WtOp.OP_TYPE = O.OP_TYPE -- Join to get workstation types based on operation type
        WHERE O.ProductID = product_id -- Filter for the specified product
        GROUP BY O.OPID, WtOp.WTID -- Group by operation ID and workstation type ID

        UNION ALL

        -- Second query: Operations related to the product's output parts
        SELECT O.OPID, WtOp.WTID -- Operation ID and Workstation Type ID
        FROM Operation O
        JOIN BOO_OUTPUT BO ON O.OPID = BO.OPID -- Join to get operations linked to output parts
        JOIN WorkstationTypes_Operation_Type WtOp ON WtOp.OP_TYPE = O.OP_TYPE -- Join to get workstation types
        WHERE BO.PartNumber IN ( -- Filter for output parts associated with the product
            SELECT PartNumber
            FROM BOO_INPUT
            WHERE OPID IN ( -- Find input operations related to the product
                SELECT OPID
                FROM Operation
                WHERE ProductID = product_id -- Filter by the product ID
            )
        )
        GROUP BY O.OPID, WtOp.WTID; -- Group by operation ID and workstation type ID

    -- Return the cursor with the results
    RETURN operation_workstationTypes_cursor;
END;
/

------- USBD14 - As a Plant Manager, I want to know which product uses all types of machines available in the factory.

CREATE OR REPLACE FUNCTION list_product_haveALL_operations
RETURN SYS_REFCURSOR -- Returns a cursor with the products using all workstations
AS
    product_haveAll_operations_cursor SYS_REFCURSOR; -- Declare cursor to hold the result
    total_workstations_count INTEGER; -- Variable to hold the total count of distinct workstation types
BEGIN
    -- Count the total number of distinct workstation types from the WorkstationTypes table
    SELECT COUNT(DISTINCT WTID)
    INTO total_workstations_count
    FROM WorkstationTypes_Operation_Type;

    -- Open the cursor to return products that use all workstation types
    OPEN product_haveAll_operations_cursor FOR
        SELECT P.ProductID -- Return the product ID
        FROM Product P
        JOIN Operation O ON P.ProductID = O.ProductID -- Join to link products with operations
        JOIN WorkstationTypes_Operation_Type WTOT ON O.OP_TYPE = WTOT.OP_TYPE -- Join to match operations to workstation types
        JOIN WorkstationTypes WT ON WTOT.WTID = WT.WTID -- Join to get the workstation type
        GROUP BY P.ProductID -- Group by product ID
        HAVING COUNT(DISTINCT WT.WTID) = total_workstations_count; -- Ensure the product uses all workstation types

    -- Return the cursor with the results
    RETURN product_haveAll_operations_cursor;
END;

    p_order_id Orders.OrderID%TYPE,  -- Order ID type from Orders table
    p_client_id Orders.ClientID%TYPE,  -- Client ID type from Orders table
    p_order_date Orders.DateOrder%TYPE,  -- Order date type from Orders table
    p_delivery_date Orders.DateDelivery%TYPE,  -- Delivery date type from Orders table
    p_status Orders.Status%TYPE,  -- Status type from Orders table
    p_product_id Product_Orders.ProductID%TYPE,  -- Product ID type from Product_Orders table
    p_quantity Product_Orders.QuantityProducts%TYPE  -- Quantity type from Product_Orders table
)

IS
    v_exists NUMBER(1);  -- Variable to check if the client is active
    v_exists_p NUMBER(1);  -- Variable to check if the product exists
    result_message VARCHAR2(255);  -- Variable to store the result message

    -- Check if the client is active
    SELECT COUNT(*)
    INTO v_exists
    FROM Client
    WHERE ClientID = p_client_id AND Status = 'activated';

    IF v_exists = 0 THEN
        RETURN 'Error: Client with ID ' || p_client_id || ' is not active or does not exist.';

    SELECT COUNT(*)
    INTO v_exists_p
    FROM Product
    WHERE ProductID = p_product_id;

    IF v_exists_p = 0 THEN

    -- Check if the order date is valid (must be in the future)

        INSERT INTO Product_Orders (OrderID, ProductID, QuantityProducts)

        result_message := 'Success: Order with ID ' || p_order_id || ' registered successfully.';
        RETURN result_message;

    EXCEPTION
        -- Handle duplicate order ID error
        WHEN DUP_VAL_ON_INDEX THEN
            RETURN 'Error: Order with ID ' || p_order_id || ' already exists.';
        -- Catch all other errors
        WHEN OTHERS THEN
            RETURN 'Unexpected error: ' || SQLERRM;
    END;
END;

------- USBD19 -  As a Production Manager, I want to know which product has more operations in its BOO.

SELECT B.ProductID,  -- ProductID from BOO table
       COUNT(O.OPID) AS operation_count  -- Count the number of operations per ProductID
FROM BOO B
JOIN Operation O ON B.ProductID = O.ProductID  -- Join BOO with Operation based on ProductID
GROUP BY B.ProductID  -- Group by ProductID from BOO table
HAVING COUNT(O.OPID) = (  -- Filter for ProductID with the highest number of operations
    SELECT MAX(operation_count)  -- Find the maximum count of operations across all ProductIDs
    FROM (
        SELECT COUNT(OPID) AS operation_count  -- Count operations per ProductID in the subquery
        FROM Operation  -- Table representing operations in the subquery
        GROUP BY ProductID  -- Group by ProductID from the Operation table
    ) operation_counts  -- The result of the subquery, representing the operation counts
);
